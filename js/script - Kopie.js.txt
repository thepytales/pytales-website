    // === LEXIKON MODAL LOGIK ===
    window.openLexikonModal = function(term) {
        const dialog = document.getElementById('lexikon-dialog');
        const title = document.getElementById('lexikon-title');
        const content = document.getElementById('lexikon-content');
        
        if(dialog) {
            // Prüfen, ob der Begriff in unserer Datenbank existiert
            if (lexikonData[term]) {
                title.textContent = lexikonData[term].title;
                content.innerHTML = lexikonData[term].content;
            } else {
                // Fallback, falls ein Begriff noch keinen Text hat
                const formattedTerm = term.split('-').map(word => word.charAt(0).toUpperCase() + word.slice(1)).join(' ');
                title.textContent = formattedTerm;
                content.innerHTML = `<p>Der Text für <strong>${formattedTerm}</strong> wird noch hinzugefügt.</p>`;
            }
            
            dialog.showModal();
        }
    };

    window.closeLexikonModal = function() {
        const dialog = document.getElementById('lexikon-dialog');
        if(dialog) dialog.close();
    };
});

document.addEventListener('DOMContentLoaded', () => {
    const themeToggleBtn = document.getElementById('theme-toggle');
    const body = document.body;
    
    // Prüfe beim Laden, ob der Nutzer schon mal den Light Mode gewählt hat
    const currentTheme = localStorage.getItem('pytales-theme');
    
    if (currentTheme === 'light') {
        body.classList.add('light-mode');
        if (themeToggleBtn) themeToggleBtn.textContent = '🌙';
    }

    // Klick-Logik für den Button
    if (themeToggleBtn) {
        themeToggleBtn.addEventListener('click', () => {
            body.classList.toggle('light-mode');
            
            // Status prüfen und speichern
            if (body.classList.contains('light-mode')) {
                localStorage.setItem('pytales-theme', 'light');
                themeToggleBtn.textContent = '🌙';
            } else {
                localStorage.setItem('pytales-theme', 'dark');
                themeToggleBtn.textContent = '☀️';
            }
        });
    }
});
